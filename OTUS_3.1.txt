#Добавим репо angie
tee /etc/yum.repos.d/angie.repo<<EOF
[angie]
name=Angie repo
baseurl=https://download.angie.software/angie/oracle/$releasever/
gpgcheck=1
enabled=1
gpgkey=https://angie.software/keys/angie-signing.gpg.asc
EOF

#Установим пакет angie и модули
yum install -y angie angie-module-brotli angie-module-auth-ldap

#Подключим модули в контексте main{} в /etc/angie/angie.conf
load_module modules/ngx_http_brotli_filter_module.so;
load_module modules/ngx_http_auth_ldap_module.so;

#Проверим конфигурацию
angie -t

#Включим в автозагрузку и запустим angie
systemctl enable --now angie.service &&\
systemctl status angie.service &&\
echo "angie started" || echo "!!angie not started"
