##/etc/angie/angie.conf
user  angie;
worker_processes  auto;
worker_rlimit_nofile 65536;

error_log  /var/log/angie/error.log notice;
pid        /run/angie.pid;

events {
    worker_connections  65536;
}


http {

    include       /etc/angie/mime.types;
    default_type  application/octet-stream;

    log_format  main  '$remote_addr - $remote_user [$time_local] "$request" '
                      '$status $body_bytes_sent "$http_referer" '
                      '"$http_user_agent" "$http_x_forwarded_for"';

    log_format extended '$remote_addr - $remote_user [$time_local] "$request" '
                        '$status $body_bytes_sent "$http_referer" rt="$request_time" '
                        '"$http_user_agent" "$http_x_forwarded_for" '
                        'h="$host" sn="$server_name" ru="$request_uri" u="$uri" '
                        'ucs="$upstream_cache_status" ua="$upstream_addr" us="$upstream_status" '
                        'uct="$upstream_connect_time" urt="$upstream_response_time"';

    access_log  /var/log/angie/access.log  main;

    sendfile        on;
    #tcp_nopush     on;

    keepalive_timeout  65;

    #gzip  on;

    include /etc/angie/http.d/*.conf;
}

#    include /etc/angie/stream.d/*.conf;
#}

##/etc/angie/http.d/default.conf
server {
    listen       80;
    server_name  _;

    rewrite_log on;
    access_log /var/log/angie/host.access.log  main;

    set $root_path '/usr/share/angie/html';

    location / {
        root   $root_path;
        index  index.html index.htm;
    }

    location /css/
             /fonts/
             /js/
             /sass/ {
        root $root_path/assets;
    }

    location /images/ {
        root $root_path;
    }

    location ~*\.(jpg|jpeg|png|gif)$ {
        root $root_path;
    }

    rewrite ^/old-images/(.*)$ /images/$1 permanent;

}

server {
    listen 1080;
    server_name _;
    return 301 http://angie-test1$request_uri;
}
